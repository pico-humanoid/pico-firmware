cmake_minimum_required(VERSION 3.22)
project(pico_device_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(PICO_TEST_SANITIZERS "Enable address/undefined behavior sanitizers on host" OFF)
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
if(PICO_TEST_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
enable_testing()
set(APP ${CMAKE_CURRENT_SOURCE_DIR}/../App)
add_library(device_protocols STATIC ${APP}/protocol/bms_protocol.c ${APP}/protocol/power_protocol.c ${APP}/device_reader.c ${APP}/dual_device_reader.c)
target_include_directories(device_protocols PUBLIC ${APP})
add_executable(test_protocols test_protocols.c)
target_link_libraries(test_protocols PRIVATE device_protocols)
add_test(NAME device_protocols_and_scheduler COMMAND test_protocols)
add_executable(test_transport test_transport.c ${APP}/transport/can_transport.c)
target_include_directories(test_transport PRIVATE fakes ${APP})
add_test(NAME can_transport_fake_hal COMMAND test_transport)
# Also compile the actual UART/application glue against the fake HAL.
add_executable(test_app test_app.c ${APP}/device_read_app.c)
target_include_directories(test_app PRIVATE fakes ${APP})
target_compile_definitions(test_app PRIVATE PICO_DEVICE_KIND=1 PICO_DEVICE_BITRATE=1000000
    PICO_POWER_ID=170 PICO_BMS_RTR_DLC=8 PICO_BMS_CRC_ORDER=1 PICO_POWER_STATUS_ORDER=0)
target_link_libraries(test_app PRIVATE device_protocols)
add_test(NAME device_application_fake_io COMMAND test_app)
add_executable(test_app_bms test_app.c ${APP}/device_read_app.c)
target_include_directories(test_app_bms PRIVATE fakes ${APP})
target_compile_definitions(test_app_bms PRIVATE PICO_DEVICE_KIND=0 PICO_DEVICE_BITRATE=500000
    PICO_POWER_ID=170 PICO_BMS_RTR_DLC=8 PICO_BMS_CRC_ORDER=1 PICO_POWER_STATUS_ORDER=0)
target_link_libraries(test_app_bms PRIVATE device_protocols)
add_test(NAME bms_application_fake_io COMMAND test_app_bms)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME board_log_capture_pty COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_capture.py)

# Retain RAW coverage and test the actual confirmed-BIG application profile too.
add_executable(test_app_power_big test_app.c ${APP}/device_read_app.c)
target_include_directories(test_app_power_big PRIVATE fakes ${APP})
target_compile_definitions(test_app_power_big PRIVATE PICO_DEVICE_KIND=1 PICO_DEVICE_BITRATE=1000000
    PICO_POWER_ID=170 PICO_BMS_RTR_DLC=8 PICO_BMS_CRC_ORDER=1 PICO_POWER_STATUS_ORDER=1)
target_link_libraries(test_app_power_big PRIVATE device_protocols)
add_test(NAME power_big_application_fake_io COMMAND test_app_power_big)

foreach(direction IN ITEMS forward reverse)
    if(direction STREQUAL "forward")
        set(source_baud 1000000)
        set(target_baud 500000)
    else()
        set(source_baud 500000)
        set(target_baud 1000000)
    endif()
    add_executable(test_baud_${direction} test_power_baud.c ${APP}/power_baud_test.c)
    target_include_directories(test_baud_${direction} PRIVATE fakes ${APP})
    target_compile_definitions(test_baud_${direction} PRIVATE PICO_DEVICE_KIND=1 PICO_POWER_ID=170
        PICO_POWER_BAUD_TEST=1 PICO_DEVICE_BITRATE=${source_baud} PICO_POWER_BAUD_TARGET=${target_baud})
    target_link_libraries(test_baud_${direction} PRIVATE device_protocols)
    add_test(NAME power_baud_gate_${direction} COMMAND test_baud_${direction})
endforeach()
add_executable(test_baud_force test_power_baud_force.c ${APP}/power_baud_test.c)
target_include_directories(test_baud_force PRIVATE fakes ${APP})
target_compile_definitions(test_baud_force PRIVATE PICO_DEVICE_KIND=1 PICO_POWER_ID=170
    PICO_POWER_BAUD_TEST=1 PICO_DEVICE_BITRATE=1000000 PICO_POWER_BAUD_TARGET=500000
    PICO_POWER_BAUD_FORCE_OPERATOR=1 POWER_BAUD_BASELINE_MS=60000UL)
target_link_libraries(test_baud_force PRIVATE device_protocols)
add_test(NAME power_baud_force_operator_one_shot COMMAND test_baud_force)
add_executable(test_app_baud test_app.c ${APP}/device_read_app.c ${APP}/power_baud_test.c)
target_include_directories(test_app_baud PRIVATE fakes ${APP})
target_compile_definitions(test_app_baud PRIVATE PICO_DEVICE_KIND=1 PICO_DEVICE_BITRATE=1000000
    PICO_POWER_ID=170 PICO_BMS_RTR_DLC=8 PICO_BMS_CRC_ORDER=1 PICO_POWER_STATUS_ORDER=1
    PICO_POWER_BAUD_TEST=1 PICO_POWER_BAUD_TARGET=500000)
target_link_libraries(test_app_baud PRIVATE device_protocols)
add_test(NAME power_baud_application_fake_io COMMAND test_app_baud)
add_test(NAME power_capture_analysis_synthetic COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_power_capture_analysis.py)

add_executable(test_dual_reader test_dual_reader.c)
target_link_libraries(test_dual_reader PRIVATE device_protocols)
add_test(NAME dual_reader_shared_bus_scheduler COMMAND test_dual_reader)
add_executable(test_app_dual test_app_dual.c ${APP}/device_read_app.c)
target_include_directories(test_app_dual PRIVATE fakes ${APP})
target_compile_definitions(test_app_dual PRIVATE PICO_DEVICE_DUAL=1 PICO_DEVICE_KIND=0
    PICO_DEVICE_BITRATE=500000 PICO_POWER_ID=170 PICO_BMS_RTR_DLC=0
    PICO_BMS_CRC_ORDER=1 PICO_POWER_STATUS_ORDER=1)
target_link_libraries(test_app_dual PRIVATE device_protocols)
add_test(NAME dual_application_fake_io COMMAND test_app_dual)
